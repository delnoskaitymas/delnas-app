function handlePhoto(e,i){ const file=e.target.files[0]; if(!file)return; // Android fix — priiname bet koki image formata const allowedTypes=['image/jpeg','image/jpg','image/png','image/webp','image/heic','image/heif']; if(!file.type.startsWith('image/')){ document.getElementById('upload-err').textContent='Netinkamas failas. Pasirink nuotrauką.'; document.getElementById('upload-err').style.display='block'; return; } const reader=new FileReader(); reader.onload=ev=>{ const image=new Image(); image.onload=()=>{ const canvas=document.createElement('canvas'); const ctx=canvas.getContext('2d'); let w=image.width,h=image.height; const max=1200; if(w>max||h>max){ if(w>h){h=Math.round(h*max/w);w=max;} else{w=Math.round(w*max/h);h=max;} } canvas.width=w;canvas.height=h; ctx.drawImage(image,0,0,w,h); const compressed=canvas.toDataURL('image/jpeg',0.85); photos[i]={data:compressed,type:'image/jpeg'}; const slot=document.getElementById('slot'+i); let img=slot.querySelector('img'); if(!img){img=document.createElement('img');slot.appendChild(img);} img.src=compressed; slot.querySelector('.slot-ico').style.display='none'; slot.querySelector('.slot-txt').style.display='none'; let rb=slot.querySelector('.rm-btn'); if(!rb){ rb=document.createElement('button'); rb.className='rm-btn'; rb.innerHTML='×'; rb.onclick=ev2=>{ev2.preventDefault();ev2.stopPropagation();removePhoto(i);}; slot.appendChild(rb); } rb.style.display='flex'; updateCount(); }; image.onerror=()=>{ document.getElementById('upload-err').textContent='Nepavyko įkelti nuotraukos. Bandyk kitą.'; document.getElementById('upload-err').style.display='block'; }; image.src=ev.target.result; }; reader.onerror=()=>{ document.getElementById('upload-err').textContent='Klaida skaitant failą. Bandyk dar kartą.'; document.getElementById('upload-err').style.display='block'; }; reader.readAsDataURL(file); }
